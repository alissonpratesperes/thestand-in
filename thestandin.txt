	class : entity
	{
		entity | Description: "Who another entities will extend of to it."
		{
			public Guid Id { get; private set; }
			public DateTime CreatedAt { get; private set; }
			public DateTime UpdatedAt { get; private set; }
		},

		prospect | Description: "Who will attend to the requesteds Dates." | One-To-Many
		{
			[ ] Queries: List, Get
			[ ] Commands: Create, Read, Update, Delete

				public string Name { get; private set; }
				public decimal Goal { get; private set; }
				public string Avatar { get; private set; } <Upload | ExtensionMethod>
				public string WhatsApp { get; private set; }
				public string Biography { get; private set; }
				public DateOnly BirthDate { get; private set; }

					public Situation Situation | <ValueObject>
					{
						public bool Active { get; private set; }
						public bool Available { get; private set; }
					}

						private readonly List<Date> _dates = new List<Date>();
						public IReadOnlyCollection<Date> Dates => _dates;
						public void NewDate(Date date) => _dates.Add(date);
		},

		date | Description: "Who will request the Dates to the Prospects." | Many-To-One, One-To-One, One-To-Many
		{
			[ ] Queries: List, Get
			[ ] Commands: Create, Read, Update, Delete
			[ ] Actions: Request, Accept, Finish, Refuse

				public string Name { get; private set; }
				public string Title { get; private set; }
				public EStatus Status { get; private set; } <Enumerator | Extension Method>
				public string WhatsApp { get; private set; }
				public DateTime Schedule { get; private set; }
				public string Description { get; private set; }
				public decimal Contribution { get; private set; }
				public EDisplacement Displacement { get; private set; } <Enumerator | Extension Method>

					public Guid ProspectId { get; private set; }
					public Prospect Prospect { get; private set; }

					public Guid LocationId { get; private set; }
					public Location Location { get; private set; }

						private readonly List<Personality> _personalities = new List<Personality>();
						public IReadOnlyCollection<Personality> Personalities => _personalities;
						public void NewPersonality(Personality personality) => _personalities.Add(personality);
		},

		location | Description: "Will be the location of requested Dates." | One-To-One
		{
			[ ] Queries: List, Get
			[ ] Commands: Create, Read, Update, Delete

				public string Latitude { get; private set; }
				public string Longitude { get; private set; }

					public Guid DateId { get; private set; }
					public Date Date { get; private set; }
		},

		personality | Description: "Will exists inside to requested Date." | Many-To-One
		{
			[ ] Queries: List, Get
			[ ] Commands: Create, Read, Update, Delete

				public bool Funny { get; private set; }
				public bool Serious { get; private set; }
				public bool Talker { get; private set; }
				public bool Listener { get; private set; }
				public bool Outgoing { get; private set; }
				public bool Conservative { get; private set; }

					public Guid DateId { get; private set; }
					public Date Date { get; private set; }
		}
	},

	validation : restriction
	{
		[ ] prospect | Blocked when...
		{
			[ ] Update: prospect.Dates.Count(date => date.Status === EStatus.Requested) > 0
			[ ] Delete: prospect.Dates.Count > 0
		},

		[ ] date | Blocked when...
		{
			[ ] Update: date.Status !== EStatus.Requested
			[ ] Delete: date.Status !== EStatus.Requested
		}
